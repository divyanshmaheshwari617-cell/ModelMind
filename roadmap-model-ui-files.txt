    useState<LessonStage>("learn");

    activeSectionIndex,

  ] = useState(0);

  const [

  completedSectionIds,

          lesson.sections.length - 1

    );

  /*
 * Overall lesson progress must be based on
 * persisted roadmap activity state.
 *
 * Each lesson has four learning stages:
 *
 * Learn
 * Visualize
 * Code
 * Practice

 * These values come from the roadmap and

  learnCompleted,

  codeCompleted,
  practiceCompleted,

  Math.round(
    (completedLessonStages / 4) *
      100
  );

  activeSection

    ? completedSectionIds.includes(

        activeSection.id

      )

    : false;

    return;

  }



  if (activeSectionCompleted) {

    return;

  }

    ...completedSectionIds,

    activeSection.id,

  ];



  setCompletedSectionIds(

    updated

  if (

    updated.length ===

    lesson.sections.length

      ...completedSectionIds,

      activeSection.id,

      updated

  return (

              Lesson progress

              {lessonProgress}%

            </strong>

          </div>

              className="lesson-viewer-progress-fill"

              ? "lesson-stage-tab lesson-stage-tab-active"

              : "lesson-stage-tab"

          }

          onClick={() =>

            setActiveStage("learn")

                          .importantPoints
                      }
                    />
                  </>
                )}

                {parameter.commonMistakes && (
                  <>
                    <h5>
                      Common mistakes
                    </h5>

                    <TextList
                      items={
                        parameter
                          .commonMistakes
                      }
                    />
                  </>
                )}

              </article>
            )
          )}

        </div>
      )}


      {/* LEARNED ATTRIBUTES */}

      {activeSection ===
        "attributes" && (
        <div className="model-param-simple-list">

          {learnedAttributes.length ===
            0 && (
            <div className="model-param-empty">
              No learned attributes are
              documented for this model
              yet.
            </div>
          )}

          {learnedAttributes.map(
            (attribute) => (
              <article
                key={attribute.id}
                className="model-param-simple-card"
              >
                <div className="model-param-simple-title">
                  <code>
                    {attribute.name}
                  </code>

                  <h4>
                    {
                      attribute.displayName
                    }
                  </h4>
                </div>

                {attribute.type && (
                  <div className="model-param-inline-type">
                    {
                      attribute.type
                    }
                  </div>
                )}

                <p>
                  {attribute.description}
                </p>

                {attribute.interpretation && (
                  <>
                    <h5>
                      How to interpret it
                    </h5>

                    <TextList
                      items={
                        attribute
                          .interpretation
                      }
                    />
                  </>
                )}

                {attribute.importantPoints && (
                  <>
                    <h5>
                      Important points
                    </h5>

                    <TextList
                      items={
                        attribute
                          .importantPoints
                      }
                    />
                  </>
                )}

              </article>
            )
          )}

        </div>
      )}

    </section>
  );
}
